cmake_minimum_required(VERSION 3.20)
project(horisub_whisper C CXX)
set(CMAKE_CXX_STANDARD 17)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${WHISPER_CPP_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Pass -DWHISPER_CPP_DIR to pin the whisper.cpp checkout.")
endif()
add_subdirectory("${WHISPER_CPP_DIR}" whisper-build)
add_executable(horisub-whisper engine.cpp)
target_link_libraries(horisub-whisper PRIVATE whisper)
if(MINGW)
  target_link_options(horisub-whisper PRIVATE -municode)
endif()
install(TARGETS horisub-whisper RUNTIME DESTINATION .)
